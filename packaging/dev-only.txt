# --- инструменты разработчика/упаковки (в .ipk и архив не нужны) ---
tools/wsl-build.sh          — кросс-сборка в WSL (инструмент разработчика)
tools/build-mips.sh         — сборка под MIPS (инструмент разработчика)
tools/make-ipk.sh           — сборка .ipk-пакетов (инструмент упаковки)
tools/stage-release.sh      — раскладка tar-архива релиза по манифесту
tools/check-shipping.sh     — сверка манифеста с .ipk
tools/collect-log.sh        — снятие лога агента на домашний сервер (cron) для
                              датасета автообучения; на роутере не запускается
tools/dataset.py            — склейка логов в CSV (признаки TRC + исход) для
                              офлайн-обучения; на устройстве не выполняется
tools/docker-release.sh  — релизная сборка в Docker (бинарь + tar + .ipk +
                           SHA256SUMS); инструмент упаковки, на устройстве не нужен
tools/check-lang.sh         — сверка языковых словарей www/lang (dev)
tools/extract-slog.py       — извлечение шаблонов slogf() (dev)
tools/gen-lang.py           — сборка src/lang_catalog.c из lang/slog.json (build)
tools/split-lang.py         — раскладка lang/slog.json на RU/EN/ZH.json (dev)
tools/make-lang-skeleton.py — каркас словаря из извлечённых строк (dev)
tools/dev-web.py            — локальный сервер www с мок-API для проверки UI (dev)

# --- внешний наблюдатель "интерфейс -> Xray" (штатный путь — кандидат tproxy) ---
tools/egress-failover.sh    — два механизма одновременно конфликтуют; в движке
                              есть кандидат tproxy, наблюдатель оставлен
                              dev-инструментом (исходники + тест в репозитории)

# --- отладочные инструменты периода разбора 0.4.5/0.4.6 (в пакет не кладём) ---
tools/diag-watch.sh         — сборщик телеметрии (разбор OOM/reprov); вызывается
                              вручную из репозитория, пользователям не нужен
                              (soak.sh + report.sh закрывают штатную диагностику)
init/entware/S95xray-guard  — сторож единственности Xray (0.4.6): лечил следствие
                              блокирующего watchdog; с неблокирующим запуском в
                              engine.c не нужен, при этом конфликтует с агентом
                              (поднимал Xray при остановленном агенте, не
                              останавливался при удалении пакета)
